lemma Gateway_Recentness:
  all-traces
  "∀ G Q n r #i.
    (GwAccept( G, Q, n, r ) @ #i) ⇒
    ((∃ D #j #s.
       ((((DeviceSignedBeacon( D, G, n, r ) @ #j) ∧ (EpochStart( G, r ) @ #s)) ∧
         (#s < #j)) ∧
        (#j < #i)) ∧
       (¬(∃ #e. (EpochEnd( G, r ) @ #e) ∧ (#e < #i)))) ∨
     (∃ #rv. (RevealedKey( Q ) @ #rv) ∧ (#rv < #i)))"
/*
guarded formula characterizing all counter-examples:
"∃ G Q n r #i.
  (GwAccept( G, Q, n, r ) @ #i)
 ∧
  (∀ D #j #s.
    (DeviceSignedBeacon( D, G, n, r ) @ #j) ∧ (EpochStart( G, r ) @ #s)
   ⇒
    ((¬(#s < #j)) ∨
     (¬(#j < #i)) ∨
     (∃ #e. (EpochEnd( G, r ) @ #e) ∧ #e < #i))) ∧
  (∀ #rv. (RevealedKey( Q ) @ #rv) ⇒ ¬(#rv < #i))"
*/
by sorry
